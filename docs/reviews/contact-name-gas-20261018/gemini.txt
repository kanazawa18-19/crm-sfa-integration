1. セキュリティ (Security)

該当箇所: src/contact_name_job/auth.py


TXT

Python
claims = verify_oauth2_token(authorization[7:], lambda **kw: request(**{**kw, 'timeout': 10}),
                            audience=audience, clock_skew_in_seconds=30)


問題点:
Google Auth ライブラリの内部（google.auth.jwt.decode）では公開鍵証明書取得時に request(certs_url, method="GET") のように第1引数を位置引数として渡します。定義されているラムダ式はキーワード専用（**kw）となっているため、実行時に必ず TypeError: <lambda>() takes 0 positional arguments but 1 was given が発生し、例外ハンドリングによってすべての正規リクエストが 401 Unauthorized と判定されます。
TXT

最小修正:
位置引数を受け取れるように変更します。
TXT

Python
claims = verify_oauth2_token(
    authorization[7:],
    lambda *args, **kw: request(*args, **{**kw, 'timeout': 10}),
    audience=audience,
    clock_skew_in_seconds=30,
)

def import_plan(self, entries, verified):
    digest = validate_plan(entries)
    if not set(verified) <= {e['id'] for e in entries}:
        raise ValueError('計画外の保存確認IDがあります')
    with connect() as conn:
        conn.execute('SET statement_timeout = 300000')  # 5分に緩和
        with conn.transaction():
            conn.execute('SELECT pg_advisory_xact_lock(%s)', (lock_key('contact_name_job', 'import'),))
            old = conn.execute('SELECT "planHash" FROM "ContactNameJob" WHERE id=%s', (JOB_ID,)).fetchone()
            if old:
                if old['planHash'] != digest:
                    raise ValueError('既存計画との差替えはできません')
                return
            conn.execute('INSERT INTO "ContactNameJob" (id,"planHash") VALUES (%s,%s)', (JOB_ID, digest))
            chunk_size = 1000
            for start in range(0, len(entries), chunk_size):
                chunk = entries[start:start + chunk_size]
                with conn.cursor() as cur:
                    cur.executemany('''INSERT INTO "ContactNameJobItem"
                        ("jobId","notionId",ordinal,shard,plan,state) VALUES (%s,%s,%s,%s,%s,%s)''',
                        [(JOB_ID, e['id'], start + i, (start + i) % 4, Jsonb(e),
                          'verified' if e['id'] in verified else 'pending')
                         for i, e in enumerate(chunk)])

【WARN】repository.py: アドバイザリロックの明示的な解放欠落

該当箇所: src/contact_name_job/repository.py


TXT

Python
@contextmanager
def shard(self, number):
    if number not in range(4):
        raise ValueError('処理番号が不正です')
    with connect() as conn:
        acquired = conn.execute('SELECT pg_try_advisory_lock(%s) AS locked',
                                (lock_key('contact_name_job', f'{JOB_ID}:{number}'),)).fetchone()['locked']
        yield conn if acquired else None


問題点:
pg_try_advisory_lock はセッションレベルのロックです。connect() は直接接続のため接続終了時に破棄されますが、コネクション切断の遅延や将来的なプーリング導入時にロックが保持され続け、次回以降の実行がすべて busy でスキップされる恐れがあります。
TXT
他 1 件

最小修正:
コンテキスト終了時に明示的にアンロックを試みます。
TXT

Python
@contextmanager
def shard(self, number):
    if number not in range(4):
        raise ValueError('処理番号が不正です')
    key = lock_key('contact_name_job', f'{JOB_ID}:{number}')
    with connect() as conn:
        acquired = conn.execute('SELECT pg_try_advisory_lock(%s) AS locked', (key,)).fetchone()['locked']
        try:
            yield conn if acquired else None
        finally:
            if acquired:
                conn.execute('SELECT pg_advisory_unlock(%s)', (key,))
