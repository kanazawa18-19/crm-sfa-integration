def internal_domains() -> set[str]:
    """既存の取得設定を保持し、照合だけの追加社内ドメインを除外する。"""
    from src.gmail_sync.sync import internal_domains_from_env
    extra = os.environ.get("EMAIL_REVIEW_INTERNAL_DOMAINS", "")
    return set(internal_domains_from_env()) | {value.strip().lower() for value in extra.split(",") if value.strip()}


def safe_capture(message, owner: str) -> bool:
    try:
        capture(message, owner)
        return True
    except Exception as exc:
        logger.warning("メール照合受付の保存失敗: id=%s type=%s", message.id, type(exc).__name__)
        return False


def capture(message, owner: str, *, repo=None) -> None:
    if not enabled():
        return
    # 開始時刻が無ければ既存メールの大量登録を防いで設定エラーとする。
    start = datetime.fromisoformat(os.environ["EMAIL_REVIEW_START_AT"].replace("Z", "+00:00"))
    if start.tzinfo is None:
        raise ValueError("EMAIL_REVIEW_START_ATにはタイムゾーンを指定してください")
    data = asdict(message)
    if domain.authentication_notice(data):
        return
    stamp = data.get("internal_date_ms")
    if not stamp or datetime.fromtimestamp(int(stamp) / 1000, timezone.utc) < start:
        return
    owner = owner.lower()
    direction, people = domain.participants(data, owner, internal_domains())
    if not people:
        return
    body = current_body(data.get("body_text") or "")
    data["body_text"] = body if len(body) <= 20000 else ""
    review_id = domain.identity(data, owner)
    row = {"id": review_id, "owner": owner, "thread_id": data.get("thread_id"), "message": data,
           "activity_owner": owner,
           "canonical_key": review_id if data.get("message_id_header") else None,
           "subject": data.get("subject") or "件名なし", "participants": people, "direction": direction,
           "state": "new" if direction == "outbound" else "excluded",
           "delivery": "not_started", "revision": 0,
           "reason": "照合処理待ち" if direction == "outbound" else domain.OUTBOUND_ONLY_REASON,
           "candidates": [], "contacts": [], "evidence": []}
    (repo or Repository()).capture(row, owner, message.id, message.thread_id)


class Service:
    def __init__(self, repo=None, gateway=None):
        self.repo = repo or Repository()
        self.gateway = gateway or Gateway()

    def evaluate(self, row: dict) -> dict:
        if row.get("direction") != "outbound":
            return self.repo.finish(row, state="excluded", reason=domain.OUTBOUND_ONLY_REASON)
        if row.get("action_id"):
            # 再配送では本人が確定した施設・案件・照合方法を保持する。
            return self.repo.finish(row, state="approved", delivery="queued", reason="登録済みの選択を保持して再配送します")
        if row.get("attempted_properties"):
            return self.repo.finish(row, state="unknown", reason="作成試行済みです。既存登録を照合してください")
        people = row["participants"]
        contacts = self.gateway.contacts([p["email"] for p in people])
        result = domain.match_existing_contact(row["message"], people, contacts)
        result["contacts"] = contacts
        if not row.get("canonical_key"):
            result.update(state="pending", reason="メール共通IDがありません。本文取得の設定後に再取得してください")
        if result["state"] == "approved":
            selection = self.gateway.validate({**row, **result, "channel": "automatic"},
                {key: result[key] for key in ("company_id", "contact_id", "target_id", "project_id")})
            result.update(selection, actor="system", channel="automatic", delivery="queued")
            result["candidates"] = [{"id": result["company_id"], "name": selection["company_name"], "sources": [], "evidence": []}]
        return self.repo.finish(row, **result)

    def decision(self, review_id: str, actor_id: str, revision: int, action: str, selection: dict, channel="dashboard") -> dict:
        row = self.repo.get(review_id, actor_id)
        # 重い外部照合より前にも版と状態を検査。保存時にもう一度原子的に検査する。
        domain.decide(row, revision=revision, action=action, selection=selection, actor=actor_id, channel=channel)
        if action == "confirm":
            if not row.get("canonical_key"):
                raise domain.Conflict("メール共通IDの取得が必要です")
            from uuid import UUID
            selection = {key: str(UUID(value)) if value else None for key, value in selection.items()}
            if selection["contact_id"] not in {c["id"] for c in row.get("contacts", [])}:
                raise domain.Conflict("連絡先候補を再照合してください")
        # 外部APIの実物照合は処理予約後に行う。Slackの受付応答を待たせない。
        return self.repo.decision(review_id, actor_id, revision, action, selection, channel)

    def process(self, row: dict) -> dict:
        if row.get("direction") != "outbound":
            # 旧版の予約も外部書込み前に止める。作成結果不明の追跡は残す。
            state = "unknown" if row.get("attempted_properties") and not row.get("action_id") else "held" if row.get("action_id") else "excluded"
            return self.repo.finish(row, state=state, reason=domain.OUTBOUND_ONLY_REASON)
        try:
            if row.get("attempted_properties") and not row.get("action_id"):
                return self.repo.finish(row, state="unknown", reason="作成試行済みです。既存登録を照合してください")
            validated = self.gateway.validate(row, row)
            actor_id = self.gateway.actor(row.get("activity_owner") or row["owner"])
            row = self.repo.checkpoint(row, **{k: validated[k] for k in ("company_name", "contact_name", "target_name", "project_name")}, actor_notion_id=actor_id)
            props = self.gateway.action_properties(row)
            contact_changes, owned_link = self.gateway.link_contact(row)
            row = self.repo.checkpoint(row, linked_company_id=row["company_id"] if owned_link else None,
                                       linked_contact_id=row["contact_id"] if owned_link else None)
            if not row.get("action_id"):
                # claim(processing) が作成予約。POST後に停止しても再作成しない。
                row = self.repo.checkpoint(row, attempted_properties=props)
                page_id = self.gateway.create_action(props)
                row = self.repo.checkpoint(row, action_id=page_id, written_properties=props)
            else:
                self.gateway.update_action(row, props)
                row = self.repo.checkpoint(row, written_properties=props)
            contact_state = self.gateway.dispatch("contact", row["contact_id"], contact_changes)
            action_state = self.gateway.dispatch("action", row["action_id"], props)
            return self.repo.finish(row, state="done", delivery="delivered" if contact_state == action_state == "delivered" else "partial",
                                    reason="記録作成済み", delivery_details={"contact": contact_state, "action": action_state})
        except domain.Conflict as exc:
            return self.repo.finish(row, state="pending", reason=str(exc), delivery="partial" if row.get("action_id") else "not_started")
        except RecordSyncBusy:
            return self.repo.finish(row, state="approved", reason="同じ対象の処理が終わってから再開します")
        except Exception as exc:
            # 例外本文はメール・認証情報を含み得るので保存しない。
            logger.warning("メール照合の外部処理失敗: id=%s type=%s", row["id"], type(exc).__name__)
            if row.get("action_id"):
                return self.repo.finish(row, state="done", delivery="partial", reason="記録作成済み・配送未完了。関連先を確認して再受付できます")
            unknown = bool(row.get("attempted_properties"))
            return self.repo.finish(row, state="unknown" if unknown else "pending", delivery="unknown" if unknown else "not_started",
                                    reason="登録結果を確認してください。再登録は行いません" if unknown else "処理に失敗しました。関連先を確認して再受付してください")

    def reconcile(self, row: dict) -> dict:
        try:
            page_id = row.get("action_id") or self.gateway.find_action(row)
            if page_id:
                return self.repo.finish(row, action_id=page_id, written_properties=row.get("attempted_properties") or row.get("written_properties"),
                                        state="approved", reason="既存の登録を確認。同じ記録の配送を再開します")
        except Exception:
            pass
        return self.repo.finish(row, state="unknown", reason="既存登録を一意に確認できません。管理者による原本照合が必要です。再登録は行いません")

    def drain(self, limit=5) -> dict:
        if not enabled():
            return {"enabled": False, "evaluated": 0, "processed": 0}
        self.repo.recover_stale()
        evaluated = processed = 0
        for _ in range(limit):
            row = self.repo.claim("recheck", "reconciling")
            if not row:
                break
            self.reconcile(row)
        for _ in range(limit):
            row = self.repo.claim("new", "evaluating")
            if not row:
                break
            try:
                self.evaluate(row)
            except Exception:
                self.repo.finish(row, state="pending", reason="候補の読込みに失敗しました。再照合してください")
            evaluated += 1
        for _ in range(limit):
            row = self.repo.claim("approved", "processing")
            if not row:
                break
            try:
                self.process(row)
            except domain.Conflict:
                logger.warning("メール照合の処理権が更新されました: id=%s", row["id"])
            processed += 1
        return {"enabled": True, "evaluated": evaluated, "processed": processed}

@contextmanager
def record_lock(db_key: str, key: str):
    conn = _connect_direct()
    try:
        conn.autocommit = True
        if not conn.execute('SELECT pg_try_advisory_lock(%s) AS locked', (lock_key(db_key, key),)).fetchone()["locked"]:
            raise RecordSyncBusy("同じ対象を処理中です")
        yield
    finally:
        conn.close()


class Gateway:
    def client(self, key: str):
        schema = get_schema(key)
        return HttpNotionClient(key, schema.notion_database_id)

    def live(self, key: str, page_id: str) -> dict:
        from src.record_merge.aliases import resolve_alias, require_record_available
        page_id = str(UUID(page_id))
        if resolve_alias(key, Tool.NOTION, page_id):
            raise Conflict("統合済みの候補です。正本を選び直してください")
        require_record_available(key, page_id)
        client = self.client(key)
        schema = get_schema(key)
        raw = client.get_raw_page_with_relations(page_id, {p.name for p in schema.properties if p.relation_target})
        if not raw or raw.get("archived") or raw.get("in_trash"):
            raise Conflict("候補が削除・保管されています")
        parent = raw.get("parent") or {}
        if str(parent.get("database_id", "")).replace("-", "") != schema.notion_database_id.replace("-", ""):
            raise Conflict("候補の種類が違います")
        result = {}
        for name, value in (raw.get("properties") or {}).items():
            if value.get("type") in {"title", "rich_text", "relation", "email", "select", "date", "phone_number"}:
                result[name] = parse_notion_property_value(value)
        return result

    def contacts(self, addresses: list[str]) -> list[dict]:
        result = []
        client = self.client("contact")
        for address in addresses:
            pages = client.query_all_pages(filter={"property": "メールアドレス", "email": {"equals": address}})
            for page in pages:
                if page.get("archived") or page.get("in_trash"):
                    continue
                props = page.get("properties") or {}
                email = parse_notion_property_value(props["メールアドレス"])
                if not isinstance(email, str) or email.casefold() != address.casefold():
                    continue
                current = self.live("contact", page["id"])
                if str(current.get("メールアドレス") or "").casefold() != address.casefold():
                    continue
                result.append({"id": page["id"], "name": current.get("名前") or address,
                               "email": address, "affiliations": current.get("取引先マスター") or []})
        return list({item["id"]: item for item in result}.values())

    def validate(self, row: dict, selection: dict) -> dict:
        company_id = selection["company_id"]
        company = self.live("client_master", company_id)
        email_match = row.get("channel") == "automatic" and row.get("match_basis") == "contact_email"
        if email_match:
            # 評価後にメール・所属・同一メールの件数が変わっていないか、作成直前にも確認する。
            current = match_existing_contact(row["message"], row["participants"],
                self.contacts([p["email"] for p in row["participants"]]))
            if current["state"] != "approved" or any(current[k] != selection.get(k) for k in ("company_id", "contact_id", "target_id", "project_id")):
                raise Conflict("メールアドレスまたは連絡先の所属が変わっています。再照合してください")
        elif row.get("channel") == "automatic":
            chosen = next((c for c in row.get("candidates", []) if c["id"] == company_id), None)
            if not chosen or normalized(chosen["name"]) != normalized(company.get("取引先名") or ""):
                raise Conflict("会社名が変わっています。再照合してください")
        contact = self.live("contact", selection["contact_id"])
        addresses = {p["email"] for p in row["participants"]}
        if str(contact.get("メールアドレス") or "").lower() not in addresses:
            raise Conflict("メールの相手と連絡先が一致しません")
        # 手動確定でも別会社への所属付け替えを暗黙に行わない。
        affiliations = contact.get("取引先マスター") or []
        if email_match and set(affiliations) != {company_id}:
            raise Conflict("連絡先の所属が変わっています。再照合してください")
        owned_correction = (row.get("channel") in {"dashboard", "slack"}
                            and row.get("linked_contact_id") == selection["contact_id"]
                            and set(affiliations) == {row.get("linked_company_id")})
        if affiliations and set(affiliations) != {company_id} and not owned_correction:
            raise Conflict("連絡先の既存所属と異なります。所属を確認してください")
        target = selection.get("target_id") or company_id
        target_company = self.live("client_master", target)
        project_name = None
        if selection.get("project_id"):
            project = self.live("project", selection["project_id"])
            if set(project.get("取引先マスター") or []) != {target}:
                raise Conflict("案件と対象施設の組合せが変わっています")
            project_name = project.get("案件名") or "名称なし"
        return {**selection, "company_name": company.get("取引先名") or "名称なし", "contact_name": contact.get("名前") or "名前未設定",
                "target_name": target_company.get("取引先名") or "名称なし", "project_name": project_name}

    def actor(self, owner: str) -> str:
        from src.sync_engine.owner_mapping import owner_directory
        directory = owner_directory()
        matches = [r for r in (directory or {}).get("users", []) if str(r.get("email") or "").lower() == owner.lower()]
        if len(matches) != 1:
            raise Conflict("メール担当者とCRMユーザーの確認済み対応を設定してください")
        user_id = str(UUID(matches[0]["notion_id"]))
        user = self.client("action").get_user(user_id)
        if user.get("type") != "person" or str((user.get("person") or {}).get("email") or "").lower() != owner.lower():
            raise Conflict("メール担当者とNotionユーザーが一致しません")
        return user_id

    def link_contact(self, row: dict) -> tuple[dict, bool]:
        with record_lock("contact", row["contact_id"]):
            self.validate(row, row)
            current = self.live("contact", row["contact_id"])
            changed = set(current.get("取引先マスター") or []) != {row["company_id"]}
            if changed:
                self.client("contact").update_page(row["contact_id"], {"取引先マスター": [row["company_id"]]})
        owned = changed or (row.get("linked_contact_id") == row["contact_id"] and row.get("linked_company_id") == row["company_id"])
        return {"取引先マスター": [row["company_id"]]}, owned

    def action_properties(self, row: dict) -> dict:
        require_outbound(row)
        message = row["message"]
        body = current_body(message.get("body_text") or "").replace("\r\n", "\n")
        if not isinstance(body, str) or not body.strip():
            raise Conflict("メール本文の全文を取得できないため、記録を保留します。元メールを確認してください")
        if len(body) > 20000:
            raise Conflict("引用を除いた本文が2万文字を超えるため、記録を保留します")
        # Notionの保存精度（分単位・ミリ秒表記）に揃える。元の送信時刻は受付に保持する。
        day = datetime.fromtimestamp(int(message["internal_date_ms"]) / 1000, tz=ZoneInfo("Asia/Tokyo")).replace(second=0, microsecond=0).isoformat(timespec="milliseconds")
        title = f"【メール】{message.get('subject') or '件名なし'} #mail-{row['id'][:12]}"
        return {"商談回数・電話回数・メール回数（何回目）": title[:180],
                "アクション種別": "メール",
                "アクション日": day, "先方担当者": row["contact_name"],
                "👨‍👩‍👧‍👦 取引先マスター": [row.get("target_id") or row["company_id"]],
                "案件名": [row["project_id"]] if row.get("project_id") else [],
                "履歴メモ": body}

    def create_action(self, properties: dict) -> str:
        return self.client("action").create_page_once(properties)

    def find_action(self, row: dict) -> str | None:
        expected = row.get("attempted_properties") or {}
        title_key = "商談回数・電話回数・メール回数（何回目）"
        if not expected.get(title_key):
            return None
        pages = self.client("action").query_all_pages(filter={"property": title_key, "title": {"equals": expected[title_key]}})
        ids = [p["id"] for p in pages if not p.get("archived") and not p.get("in_trash")]
        if len(ids) != 1:
            return None
        current = self.live("action", ids[0])
        if any(not same_value(key, current.get(key), value) for key, value in expected.items()):
            return None
        return ids[0]

    def update_action(self, row: dict, properties: dict) -> None:
        with record_lock("action", row["action_id"]):
            current = self.live("action", row["action_id"])
            previous = row.get("written_properties") or {}
            if any(not same_value(key, current.get(key), previous.get(key)) and not same_value(key, current.get(key), value) for key, value in properties.items()):
                raise Conflict("記録が外部で編集されています。内容を比較してください")
            self.client("action").update_page(row["action_id"], properties)

    def dispatch(self, key: str, page_id: str, properties: dict) -> str:
        from src.sync_engine.production_wiring import get_production_wiring
        raw = self.client(key).get_raw_page(page_id)
        occurred = datetime.fromisoformat(raw["last_edited_time"].replace("Z", "+00:00"))
        event = SyncEvent(source_tool=Tool.NOTION, db_key=key, external_id=page_id, occurred_at=occurred,
                          properties=properties, expected_source_properties=properties)
        result = get_production_wiring().dispatcher.dispatch(event)
        return "partial" if result.skipped or result.has_partial_skips else "delivered"
